def to_twos_complement(value, bits=8):
    """Convert a decimal integer to its N-bit 2's complement binary string."""
    min_val = -(1 << (bits - 1))
    max_val = (1 << (bits - 1)) - 1
    if not (min_val <= value <= max_val):
        raise ValueError(f"Value {value} is out of range for {bits}-bit signed integers.")
    if value < 0:
        value = (1 << bits) + value
    return format(value, f'0{bits}b')

def from_twos_complement(binary_str, bits=8):
    """Convert an N-bit 2's complement binary string back to a decimal integer."""
    value = int(binary_str, 2)
    if value & (1 << (bits - 1)):
        value -= (1 << bits)
    return value

def perform_signed_addition(a, b, bits=8):
    """Perform signed binary addition and check for arithmetic overflow."""
    min_val = -(1 << (bits - 1))
    max_val = (1 << (bits - 1)) - 1
    
    bin_a = to_twos_complement(a, bits)
    bin_b = to_twos_complement(b, bits)
    
    raw_sum = a + b
    
    # Overflow occurs if two operands of the same sign yield a result with a different sign
    overflow = not (min_val <= raw_sum <= max_val)
    
    # Wrap-around sum for binary representation within the bit-width
    wrapped_sum = raw_sum & ((1 << bits) - 1)
    bin_res = format(wrapped_sum, f'0{bits}b')
    
    return {
        "operand_a_dec": a,
        "operand_a_bin": bin_a,
        "operand_b_dec": b,
        "operand_b_bin": bin_b,
        "result_dec": raw_sum if not overflow else from_twos_complement(bin_res, bits),
        "result_bin": bin_res,
        "overflow": overflow
    }

# --- Verification of Expressions ---
expressions = [
    (24, 17, "a) (24 + (+17))"),
    (-25, -18, "b) ((-25) + (-18))")
]

for a, b, label in expressions:
    res = perform_signed_addition(a, b, bits=8)
    print(f"Expression: {label}")
    print(f"  Operand A: {res['operand_a_dec']} = {res['operand_a_bin']}")
    print(f"  Operand B: {res['operand_b_dec']} = {res['operand_b_bin']}")
    print(f"  Result Binary: {res['result_bin']}")
    print(f"  Result Decimal: {res['result_dec']}")
    print(f"  Overflow Detected: {res['overflow']}\n")

def logical_left_shift(bin_str, shift_amount, bits=8):
    """Perform a logical left shift and display step-by-step."""
    # Ensure it fits in 'bits' by taking the lower 'bits' if longer
    if len(bin_str) > bits:
        bin_str = bin_str[-bits:]
    elif len(bin_str) < bits:
        bin_str = bin_str.zfill(bits)
        
    print(f"Original {bits}-bit binary: {bin_str}")
    val = int(bin_str, 2)
    
    current_val = val
    for i in range(1, shift_amount + 1):
        # Logical left shift: multiply by 2, mask to keep within 'bits' width
        current_val = (current_val << 1) & ((1 << bits) - 1)
        print(f"  Step {i} (Shift Left): {format(current_val, f'0{bits}b')}")
        
    return format(current_val, f'0{bits}b')

def logical_right_shift(bin_str, shift_amount, bits=8):
    """Perform a logical right shift and display step-by-step."""
    if len(bin_str) > bits:
        bin_str = bin_str[-bits:]
    elif len(bin_str) < bits:
        bin_str = bin_str.zfill(bits)
        
    print(f"Original {bits}-bit binary: {bin_str}")
    val = int(bin_str, 2)
    
    current_val = val
    for i in range(1, shift_amount + 1):
        # Logical right shift: integer division by 2
        current_val = current_val >> 1
        print(f"  Step {i} (Shift Right): {format(current_val, f'0{bits}b')}")
        
    return format(current_val, f'0{bits}b')

res_a = logical_left_shift("1111110110", 2, bits=8)

res_b = logical_right_shift("10010100", 2, bits=8)
